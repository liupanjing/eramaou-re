; 实际调用事件入口，验证新口上不改变战斗、城务或关系结果。
@EV_RUN
#DIM DYNAMIC EV_I
#DIM DYNAMIC EV_ID
#DIM DYNAMIC EV_PHASE
#DIM DYNAMIC EV_KIND
#DIM DYNAMIC EV_FLAG
#DIM DYNAMIC EV_CYCLE
#DIMS DYNAMIC EV_BEFORE
EV_FLAG = FLAG:7
FLAG:7 = 2
FOR EV_I, 0, 29
	EV_ID = DT_VOICE_ID(EV_I)
	CALL EV_PROFILE(EV_ID)
	FOR EV_PHASE, 0, 4
		CASTLE_PERSON:2:CP_ROLE = EV_PHASE == 3 ? CR_FOLLOWER # CR_CAPTIVE
		DIALOGUE_PERSON:2:DP_CHAPTER = EV_PHASE == 0 ? 0 # EV_PHASE
		DIALOGUE_PERSON:2:DP_TRUST = EV_PHASE == 1 ? 20 # 60
		DIALOGUE_PERSON:2:DP_AGREE = DIALOGUE_PERSON:2:DP_TRUST
		FOR EV_KIND, 1, 13
			SIF EV_KIND >= CE_JOIN && EV_KIND <= CE_JOB_CLEAR && EV_PHASE != 3
				CONTINUE
			PRINTFORML EVENT SAMPLE profile={EV_ID} phase={EV_PHASE} kind={EV_KIND}
			CALL CT_ASSERT(STRLENS(CHARACTER_EVENT_TEXT(2, EV_KIND, 0)) > 0 && STRLENS(CHARACTER_EVENT_TEXT(2, EV_KIND, 1)) > 0, "all event profiles provide reachable text and variants")
			EV_BEFORE '= EV_SNAPSHOT(2)
			EV_CYCLE = CHARACTER_LINE_CYCLE:EV_ID:EV_KIND
			CALL CHARACTER_EVENT(2, EV_KIND)
			CALL CT_ASSERT(RESULT == 1, "event renderer executes native output")
			CALL CHARACTER_EVENT(2, EV_KIND)
			CALL CT_ASSERT(RESULT == 1 && CHARACTER_LINE_CYCLE:EV_ID:EV_KIND == (EV_CYCLE + 2) % 10000, "event text rotates without random draws")
			CALL CT_ASSERT(EV_BEFORE == EV_SNAPSHOT(2), "event text leaves actor target money actions history and ending flags unchanged")
		NEXT
	NEXT
NEXT
; 归顺低信任仍保留本人语气；不以同一个亲近变体覆盖。
CALL EV_PROFILE(170)
CASTLE_PERSON:2:CP_ROLE = CR_FOLLOWER
DIALOGUE_PERSON:2:DP_TRUST = 0
DIALOGUE_PERSON:2:DP_AGREE = 0
CALL CT_ASSERT(CHARACTER_EVENT_TEXT(2, CE_HOME, 0) == CHARACTER_EVENT_TEXT(2, CE_HOME, 1), "reserved member never selects familiar event variant")
EV_BEFORE '= CHARACTER_EVENT_TEXT(2, CE_HOME, 0)
DIALOGUE_PERSON:2:DP_TRUST = 80
DIALOGUE_PERSON:2:DP_AGREE = 80
CALL CT_ASSERT(CHARACTER_EVENT_TEXT(2, CE_HOME, 0) != "" && CHARACTER_EVENT_TEXT(2, CE_HOME, 0) != EV_BEFORE, "trusted member receives the distinct open event variant")
; 实际旧引擎战斗分派，保留原TARGET契约，不执行错误角色的旧口上。
A = 2
B = 2
TARGET = 1
EV_CYCLE = CHARACTER_LINE_CYCLE:170:CE_ATTACK
CALL ATTACK_KOUJO, 2
CALL CT_ASSERT(TARGET == 2 && A == 2 && B == 2 && CHARACTER_LINE_CYCLE:170:CE_ATTACK == (EV_CYCLE + 1) % 10000, "actual attack dispatcher selects the attacking person")
CALL ATTACK_KOUJO_B
CALL CT_ASSERT(TARGET == 2 && CHARACTER_LINE_CYCLE:170:CE_ATTACK == (EV_CYCLE + 2) % 10000, "secondary attack dispatcher uses the same character registry")
EV_CYCLE = CHARACTER_LINE_CYCLE:170:CE_VICTORY
CALL VICTORY_KOUJO
CALL CT_ASSERT(TARGET == 2 && CHARACTER_LINE_CYCLE:170:CE_VICTORY == (EV_CYCLE + 1) % 10000, "battle winner speaks victory rather than captive defeat")
TARGET = 1
EV_CYCLE = CHARACTER_LINE_CYCLE:170:CE_ARRIVAL
CALL ENTERENEMY_KOUJO
CALL CT_ASSERT(TARGET == 1 && CHARACTER_LINE_CYCLE:170:CE_ARRIVAL == (EV_CYCLE + 1) % 10000, "arrival preserves the original current target")
; TARGET故意指向最后行动的队友，完整陷落文本须用显式攻略者。
EV_BEFORE '= EV_SNAPSHOT(2)
EV_CYCLE = CHARACTER_LINE_CYCLE:170:CE_VICTORY
CT_QUEUE = 0
CT_POS = 0
CT_SIZE = 1
CALL ENDING_2(2)
CALL CT_ASSERT(TARGET == 2 && CHARACTER_LINE_CYCLE:170:CE_VICTORY == (EV_CYCLE + 1) % 10000, "castle fall credits explicit expedition leader over stale teammate target")
TARGET = 1
CALL CT_ASSERT(EV_BEFORE == EV_SNAPSHOT(2), "castle fall reaction preserves all other actor resources and ending flags")
; 关闭口上、非法对象、错误阶段均不产生反应。
EV_CYCLE = CHARACTER_LINE_CYCLE:170:CE_ATTACK
FLAG:7 = 0
CALL CHARACTER_EVENT(2, CE_ATTACK)
CALL CT_ASSERT(RESULT == 0 && CHARACTER_LINE_CYCLE:170:CE_ATTACK == EV_CYCLE, "muted voices do not advance event rotation")
FLAG:7 = 2
CALL CHARACTER_EVENT(-1, CE_ATTACK)
CALL CT_ASSERT(RESULT == 0, "invalid actor is ignored safely")
CALL CHARACTER_EVENT(CHARANUM, CE_ATTACK)
CALL CT_ASSERT(RESULT == 0, "out of bounds actor is ignored safely")
CALL CHARACTER_EVENT(2, 13)
CALL CT_ASSERT(RESULT == 0, "undefined event does not fall through to adult content")
CASTLE_PERSON:2:CP_ROLE = CR_CAPTIVE
CALL CHARACTER_EVENT(2, CE_JOIN)
CALL CT_ASSERT(RESULT == 0, "unconfirmed captive does not speak as recruited")
CALL CHARACTER_EVENT(2, CE_HOME)
CALL CT_ASSERT(RESULT == 0, "captive does not claim a residence assignment")
; 年幼家属只参加生活与非性剧情内容，不进入任职、加入反应。
CALL EV_PROFILE(104)
CFLAG:2:451 = 10
CASTLE_PERSON:2:CP_ROLE = CR_FAMILY
CASTLE_PERSON:2:CP_HOME = CL_FAMILY
CASTLE_STATE:2 = CL_FAMILY
CALL CHARACTER_EVENT(2, CE_HOME)
CALL CT_ASSERT(RESULT == 1, "young family member can respond to safe housing")
CALL CHARACTER_EVENT(2, CE_JOIN)
CALL CT_ASSERT(RESULT == 0, "young character cannot be presented as adult recruit")
CALL CHARACTER_EVENT(2, CE_JOB)
CALL CT_ASSERT(RESULT == 0, "young character does not receive work reaction")
CALL CHARACTER_EVENT(2, CE_JOB_CLEAR)
CALL CT_ASSERT(RESULT == 0, "young character does not receive job release reaction")
; 真正的人事操作成功才反应，取消、重复安排、资金/行动不足不会出声。
CALL EV_PROFILE(169)
CASTLE_PERSON:2:CP_ROLE = CR_FOLLOWER
CASTLE_PERSON:2:CP_HOME = CL_QUARTERS
CASTLE_STATE:2 = CL_THRONE
EV_CYCLE = CHARACTER_LINE_CYCLE:169:CE_JOB
CT_QUEUE = 999
CT_POS = 0
CT_SIZE = 1
CALL CASTLE_ASSIGN(2, 4)
CALL CT_ASSERT(CASTLE_STATE:5 == 3 && CHARACTER_LINE_CYCLE:169:CE_JOB == EV_CYCLE, "cancelled appointment has no voice or action cost")
CT_QUEUE = CL_BARRACKS
CT_POS = 0
CALL CASTLE_ASSIGN(2, 4)
CALL CT_ASSERT(CASTLE_PERSON:2:CP_WORK == CL_BARRACKS && CASTLE_STATE:5 == 2 && CHARACTER_LINE_CYCLE:169:CE_JOB == (EV_CYCLE + 1) % 10000, "successful appointment reaches character reaction once")
CT_POS = 0
CALL CASTLE_ASSIGN(2, 4)
CALL CT_ASSERT(CASTLE_STATE:5 == 2 && CHARACTER_LINE_CYCLE:169:CE_JOB == (EV_CYCLE + 1) % 10000, "unchanged appointment produces no duplicate reaction")
CT_QUEUE = 0
CT_POS = 0
EV_CYCLE = CHARACTER_LINE_CYCLE:169:CE_JOB_CLEAR
CALL CASTLE_ASSIGN(2, 4)
CALL CT_ASSERT(CASTLE_PERSON:2:CP_WORK == 0 && CASTLE_STATE:5 == 1 && CHARACTER_LINE_CYCLE:169:CE_JOB_CLEAR == (EV_CYCLE + 1) % 10000, "job release uses its own response")
CT_QUEUE = CL_FAMILY
CT_POS = 0
EV_CYCLE = CHARACTER_LINE_CYCLE:169:CE_HOME
CALL CASTLE_ASSIGN(2, 3)
CALL CT_ASSERT(CASTLE_PERSON:2:CP_HOME == CL_FAMILY && CASTLE_STATE:5 == 0 && CHARACTER_LINE_CYCLE:169:CE_HOME == (EV_CYCLE + 1) % 10000, "successful residence change triggers one response")
CT_QUEUE = CL_QUARTERS
CT_POS = 0
CALL CASTLE_ASSIGN(2, 3)
CALL CT_ASSERT(CASTLE_PERSON:2:CP_HOME == CL_FAMILY && CHARACTER_LINE_CYCLE:169:CE_HOME == (EV_CYCLE + 1) % 10000, "exhausted actions cannot trigger false moving response")
EV_CYCLE = CHARACTER_LINE_CYCLE:167:CE_JOIN
CALL DT_ROUTE(167)
CALL CT_ASSERT(CHARACTER_LINE_CYCLE:167:CE_JOIN == (EV_CYCLE + 1) % 10000, "real personnel confirmation triggers joining reaction once")
; 结局回应只读状态，不新增角色、不授予效忠、不改写真正结局标志。
EV_BEFORE '= EV_SNAPSHOT(2)
FOR EV_KIND, CE_WORLD_END, CE_NORMAL_END + 1
	CALL CHARACTER_EPILOGUE(EV_KIND)
	CALL CT_ASSERT(EV_BEFORE == EV_SNAPSHOT(2), "epilogue preserves actual people allegiance and ending result")
NEXT
CASTLE_STATE:2 = CL_QUARTERS
EV_BEFORE '= EV_SNAPSHOT(2)
CALL CHARACTER_LOCAL_REST
CALL CT_ASSERT(EV_BEFORE == EV_SNAPSHOT(2), "rest flavor does not itself advance time or spend actions")
FLAG:7 = EV_FLAG
PRINTL CHARACTER EVENT ASSERTIONS PASSED
RETURN

@EV_PROFILE(EV_ID)
#DIM DYNAMIC EV_ID
CALL DT_RESET(0)
TALENT:2:15 = 0
IF EV_ID == 150
	CALL DT_RESET(150)
ELSEIF EV_ID == 134
	EX_TALENT:2:4 = 1
ELSEIF EV_ID >= 101 && EV_ID <= 104
	EX_TALENT:2:EV_ID = 1
ELSEIF EV_ID >= 160
	TALENT:2:EV_ID = 1
ENDIF
CALL DIALOGUE_SYNC(2)
SIF EV_ID < 7
	DIALOGUE_PERSON:2:DP_TEMPER = EV_ID
RETURN

@EV_SNAPSHOT(EV_P)
#FUNCTIONS
#DIM DYNAMIC EV_P
RETURNF @"{CHARANUM}/{MONEY}/{DAY}/{TIME}/{TARGET}/{ASSI}/{A}/{B}/{CASTLE_STATE:5}/{CASTLE_PERSON:EV_P:CP_ROLE}/{CASTLE_PERSON:EV_P:CP_HOME}/{CASTLE_PERSON:EV_P:CP_WORK}/{DIALOGUE_PERSON:EV_P:DP_TRUST}/{DIALOGUE_PERSON:EV_P:DP_AGREE}/{DIALOGUE_PERSON:EV_P:DP_CHAPTER}/{DIALOGUE_PERSON:EV_P:DP_PROMISE}/{DIALOGUE_PERSON:EV_P:DP_MET}/{CFLAG:EV_P:10}/{ABL:EV_P:10}/{MARK:EV_P:3}/{FLAG:82}/{FLAG:87}/{FLAG:89}/{FLAG:91}"
