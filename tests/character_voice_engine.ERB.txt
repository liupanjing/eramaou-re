		PRINTFORML VOICE SAMPLE profile={DT_ID} phase={DT_PHASE}

			CALL CT_ASSERT(STRLENS(DIALOGUE_CHARACTER_TEXT(2, DT_KIND)) > 1, "encounter dialogue and response text resolve for every profile")

				CALL DIALOGUE_PROSE(2, DIALOGUE_CHARACTER_TEXT(2, DT_KIND))

	NEXT
